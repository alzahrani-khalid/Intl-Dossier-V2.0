New config contract this plan defines (consumed by Wave-2 wiring plans):

```typescript
export interface FilterFieldDef {
  key: string // URL param name ('status', 'sensitivity', 'type', ...)
  labelKey: string // list-controls:… or surface ns key
  options: { value: string; labelKey: string }[]
  buildCountQuery?: (value: string, active: Record<string, string | undefined>) => Promise<number>
}
export interface SortFieldDef {
  id: string
  labelKey: string
}
export interface PropertyDef {
  id: string
  labelKey: string
  defaultVisible: boolean
}
export interface GroupingDef {
  id: string
  labelKey: string
}
export interface ListControlsConfig {
  filters: FilterFieldDef[]
  sortFields?: SortFieldDef[] // omit → no Sort by section
  properties?: PropertyDef[] // omit or < 2 entries → no Display properties section (Open Q3)
  grouping?: GroupingDef[] // kanban only; omit → 'No grouping' static row
}
```
